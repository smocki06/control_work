#include <iostream>

int main() {
    // Объявление переменных для хранения коэффициентов
    double x, y, z;

    // Ввод значений переменных с клавиатуры
    std::cout << "Введите значение x: ";
    std::cin >> x;
    std::cout << "Введите значение y: ";
    std::cin >> y;
    std::cout << "Введите значение z: ";
    std::cin >> z;

    // Вычисление значения упрощённого выражения: -10y + 14z
    // Исходное выражение: 2.5*(8x-4y) - 4*(5x-3.5z)
    double result = -10.0 * y + 14.0 * z;

    std::cout << "Значение выражения: " << result << std::endl;

    return 0;
}
